// GitHub Pages serves this one page for every 404, including /ko/* paths, so the
// language has to be picked here rather than by having a second Korean 404 page.
(function () {
  var ko = /^\/ko(\/|$)/.test(window.location.pathname);
  if (!ko) { try { ko = localStorage.getItem('site_lang') === 'ko'; } catch (e) {} }
  if (!ko) return;
  document.documentElement.lang = 'ko';
  document.title = '페이지를 찾을 수 없습니다 - {{ site.name | escape }}';
  var set = function (id, text) { var el = document.getElementById(id); if (el) el.textContent = text; };
  set('nf-msg', '찾으시는 페이지가 없습니다.');
  set('nf-home', '홈으로');
  set('nf-search', '사이트 검색');
  set('nf-try', '이런 페이지는 어떠세요');
  var links = document.querySelectorAll('a[data-ko-href]');
  for (var i = 0; i < links.length; i++) {
    links[i].href = links[i].getAttribute('data-ko-href');
    links[i].textContent = links[i].getAttribute('data-ko-label');
  }
  var toggle = document.querySelector('.lang-toggle');
  if (toggle) {
    toggle.href = '{{ site.url }}{{ site.baseurl }}/';
    toggle.textContent = 'EN';
    // The header renders this toggle for an English page, so its inline onclick
    // stores 'ko'. Left alone, clicking EN would save 'ko' and head.html would
    // redirect / straight back to /ko/.
    toggle.onclick = function () { try { localStorage.setItem('site_lang', 'en'); } catch (e) {} };
  }
})();
</script>
